package booksearchsilsoop2.controller;

import booksearchsilsoop2.service.javaFXTableViewService;
import booksearchsilsoop2.service.javaFXTableViewServiceImpl;
import booksearchsilsoop2.vo.Book;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.Button;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;

import java.net.URL;
import java.util.ResourceBundle;

public class javaFXTableViewController implements Initializable {

    @FXML private TableView<Book> tableView;
    @FXML private Button searchBtn;
    @FXML private Button deleteBtn;
    @FXML private TableColumn<Book, String> isbnCol;
    @FXML private TableColumn<Book, String> titleCol;
    @FXML private TableColumn<Book, Integer> priceCol;
    @FXML private TableColumn<Book, String> authorCol;
    @FXML private TextField searchId;

    javaFXTableViewService service = new javaFXTableViewServiceImpl();

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {

        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        isbnCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));


        tableView.setEditable(true);

        searchBtn.setOnAction(e -> {
            String search = searchId.getText();
            selectBook(search);
        });

        deleteBtn.setOnAction(e -> {
            Book book = tableView.getSelectionModel().getSelectedItem();
            deleteBook(book);
            tableView.getItems().remove(book);
        });

        isbnCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bisbn", e.getNewValue());
        });
        titleCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "btitle", e.getNewValue());
        });
        authorCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bauthor", e.getNewValue());
        });
        priceCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bprice", e.getNewValue().toString());
        });


    }

    private void selectBook(String search) {
        ObservableList<Book> books =
                service.selectBook(search);
        tableView.setItems(books);
    }

    private void deleteBook(Book book) {
        service.deleteBook(book);
    }

    private void updateBook(Book book, String column, String newValue) {
        service.updateBook(book, column, newValue);
        selectBook(searchId.getText());
    }

}

package booksearchsilsoop2.dao;

import booksearchsilsoop2.service.ConnectionMaker;
import booksearchsilsoop2.service.ConnectionMakerImpl;
import booksearchsilsoop2.vo.Book;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class javaFXTableViewDAO {

    private ConnectionMaker connectionMaker;

    public javaFXTableViewDAO(ConnectionMaker connectionMaker) {
        // 생성자 주입
        this.connectionMaker = connectionMaker;
    }

    public ObservableList<Book> select(String search) {
        Connection con = connectionMaker.getConnection();
        ObservableList<Book> books = FXCollections.observableArrayList();
        try {
            con.setAutoCommit(false); // Transaction 시작점
            String sql = "select bisbn, btitle, bauthor, bprice ";
            sql += "from book where btitle LIKE ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + search + "%");
            ResultSet rs = pstmt.executeQuery();
            while (rs.next()) {
                Book book = new Book(rs.getString("bisbn"),
                        rs.getString("btitle"),
                        rs.getInt("bprice"),
                        rs.getString("bauthor"));
                books.add(book);
            }
            rs.close();
            pstmt.close();
            con.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        return books;
    }

    public void delete(Book book) {
        Connection con = connectionMaker.getConnection();
        try {
            String isbn = book.getBisbn();
            con.setAutoCommit(false);
            String sql = "delete ";
            sql += "from book where bisbn LIKE ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, isbn);
        } catch(Exception e) {
            e.printStackTrace();
        }
    }

    public void update(Book book, String column, String newValue) {
        Connection con = connectionMaker.getConnection();
        try {
            con.setAutoCommit(false);
            String sql = "UPDATE book SET " + column + " = ? WHERE bisbn = ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, newValue);
            pstmt.setString(2, book.getBisbn());
            pstmt.executeUpdate();
            pstmt.close();
            con.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
    }
}
