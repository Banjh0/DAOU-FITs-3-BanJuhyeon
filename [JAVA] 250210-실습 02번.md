import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class Transaction {
    public static void main(String[] args) {
        Connection con = null;
        try {
            // 1. Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");
            
            // 2. Connection 연결
            String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
            String ID = "C##JAVA";
            String PW = "1234";
            con = DriverManager.getConnection(JDBC_URL, ID, PW);

            // 기본적으로 autoCommit mode로 Conncection이 생성
            // SQL구문 하나가 실행되면 DB가 Durability 속성으로 인해 변화하한다는 것.
            // 즉, 단일 SQL구문이 단일 트랜잭션.

            // 만약 transaction을 설정하려면
            con.setAutoCommit(false); // 이 순간부터 transaction 설정
/*            
            명시적으로 con.commit() 혹은 con.rollback()이 호출되면
            transaction이 종료
            추가적으로 만약 해당 구문이 나오지 않고 connection이 정상종료(con.close()) 되면,
            con.commit()으로 간주.
*/
            // 3. SQL구문 작성
            String sql = "insert into memberTbl values(?, ?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "2");
            pstmt.setString(2, "신사임당");
            
            // 4. SQL구문 실행
            int result = pstmt.executeUpdate(); // 리턴값은 영향받은 row의 수
           
            // 5. 결과처리
            if(result == 1) {
                System.out.println("잘 처리 되었어요!");
            }
//            con.rollback(); // transaction 무효화!
            con.commit(); // transaction 종료! 이번 transaction은 데이터베이스에 적용

            // 6. 리소스 해제
            pstmt.close();
            con.close();

        } catch(Exception e) {
            e.printStackTrace();
        }

    }
}
