package mybatisex.controller;

import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.Button;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;
import mybatisex.service.BookService;
import mybatisex.service.BookServiceImpl;
import mybatisex.vo.BookVO;

import java.net.URL;
import java.util.ResourceBundle;

public class BookController implements Initializable {

    @FXML private TableView<BookVO> tableView;
    @FXML private Button searchBtn;
    @FXML private Button deleteBtn;
    @FXML private TableColumn<BookVO, String> isbnCol;
    @FXML private TableColumn<BookVO, String> titleCol;
    @FXML private TableColumn<BookVO, Integer> priceCol;
    @FXML private TableColumn<BookVO, String> authorCol;
    @FXML private TextField searchId;
    @FXML private TextField insISBN;
    @FXML private TextField insTitle;
    @FXML private TextField insAuthor;
    @FXML private TextField insPrice;
    @FXML private Button insertBtn;

    BookService service = new BookServiceImpl();

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {

        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        isbnCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));

        tableView.setEditable(true);

        searchBtn.setOnAction(e -> {
            String search = searchId.getText();
            selectBook(search);
        });

        deleteBtn.setOnAction(e -> {
            BookVO book = tableView.getSelectionModel().getSelectedItem();
            deleteBook(book);
            tableView.getItems().remove(book);
        });

        insertBtn.setOnAction(e -> {
            String bisbn = insISBN.getText();
            String btitle = insTitle.getText();
            int bprice = Integer.parseInt(insPrice.getText());
            String bauthor = insAuthor.getText();
            if(!bisbn.isEmpty() && !btitle.isEmpty() && bprice > 0 && !bauthor.isEmpty()) {
                BookVO book = new BookVO(bisbn, btitle, bprice, bauthor);
                insertBook(book);
            } else {
                System.out.println("정보 부족");
            }
        });

        isbnCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bisbn", e.getNewValue());
        });
        titleCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "btitle", e.getNewValue());
        });
        authorCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bauthor", e.getNewValue());
        });
        priceCol.setOnEditCommit(e -> {
            updateBook(e.getRowValue(), "bprice", e.getNewValue().toString());
        });


    }

    private void updateBook(BookVO rowValue, String btitle, String newValue) {
        service.updateBook(rowValue, btitle, newValue);
        selectBook(searchId.getText());
    }

    private void deleteBook(BookVO book) {
        service.deleteBook(book);
    }

    private void selectBook(String search) {
        ObservableList<BookVO> books =
                service.selectBook(search);
        tableView.setItems(books);
    }

    private void insertBook(BookVO book) {
        service.insertBook(book);
        selectBook(searchId.getText());
    }
}

package mybatisex.service;

import javafx.collections.ObservableList;
import mybatisex.dao.BookDAO;
import mybatisex.mybatis.MyBatisSessionFactory;
import mybatisex.vo.BookVO;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;

import java.util.HashMap;
import java.util.Map;

public class BookServiceImpl implements BookService {

    SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();

    BookDAO dao = new BookDAO(factory);

    @Override
    public ObservableList<BookVO> selectBook(String search) {
        ObservableList<BookVO> books = null;
        // Transaction 관리 불요
        try {
            SqlSession session = factory.openSession(); // Transaction 자동
            if(!search.isEmpty()) {
                books = dao.select(session, search);
            } else {
                books = dao.selectAll(session);
            }
            session.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        return books;
    }

    @Override
    public void updateBook(BookVO book, String column, String newValue) {
        SqlSession session = factory.openSession(false); // 수동 Transaction
        try {
            Map<String, Object> params = new HashMap<>();
            params.put("bisbn", book.getBisbn());
            params.put("column", column);
            params.put("newValue", newValue);
            dao.update(session, params);
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }

    }

    @Override
    public void deleteBook(BookVO book) {
        SqlSession session = factory.openSession(false); // 수동 Transaction
        try {
            dao.delete(session, book.getBisbn());
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }
    }

    @Override
    public void insertBook(BookVO book) {
        SqlSession session = factory.openSession(false); // 수동
        try {
            dao.insert(session, book);
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }

    }

}

        <![CDATA[
            DELETE
            FROM book
            WHERE bisbn = #{bsibn}
        ]]>

            INSERT
            INTO book (bisbn, btitle, bprice, bauthor)

        ]]>

    <update id="updateBook" parameterType="Map">

            SET ${column} = #{newValue}
