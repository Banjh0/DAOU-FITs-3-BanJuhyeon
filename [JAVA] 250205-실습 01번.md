package assignment;

import java.util.*;

public class Test01 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double num1 = sc.nextDouble();
        double num2 = sc.nextDouble();
        if(num1 % num2 <= 1) {
            System.out.print("나머지가 1보다 작거나 같다!");
        } else if(num1 % num2 > 1) {
            System.out.print("나머지가 1보다 크다!");
        }
        sc.close();
    }
}
