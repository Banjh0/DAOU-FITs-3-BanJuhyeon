# 물리 모델 → 테이블 생성

## 테이블 생성한 "명령어 스크립트 파일"

```SQL
CREATE TABLE CUSTOMER_INFO (
    USER_ID NUMBER PRIMARY KEY,
    NAME VARCHAR2(30) NOT NULL,
    EMAIL VARCHAR2(50),
    PHONE NUMBER,
    ADDRESS VARCHAR2(100),
    REGISTRATION_DATE DATE DEFAULT SYSDATE
);

CREATE TABLE ACCOUNT_INFO (
    ACCOUNT_ID NUMBER PRIMARY KEY,
    USER_ID NUMBER,
    CONSTRAINT ACCOUNT_USER_ID FOREIGN KEY(USER_ID) REFERENCES CUSTOMER_INFO (USER_ID),
    ACCOUNT_NUMBER NUMBER NOT NULL,
    BALANCE NUMBER,
    CREATED_DATE DATE DEFAULT SYSDATE
);

CREATE TABLE COMPANY_INFO (
    COMPANY_ID NUMBER PRIMARY KEY,
    NAME VARCHAR2(100),
    OPEN_DATE DATE,
    CEO VARCHAR2(50)
);

CREATE TABLE STOCK_INFO (
    STOCK_ID NUMBER PRIMARY KEY,
    COMPANY_ID NUMBER,
    CONSTRAINT STOCKINFO_COMPANY_ID FOREIGN KEY(COMPANY_ID) REFERENCES COMPANY_INFO (COMPANY_ID),
    SECTOR VARCHAR2(50),
    MARKET VARCHAR2(50)
);

CREATE TABLE SECURITIES_ORDERS (
    ORDER_ID NUMBER PRIMARY KEY,
    STOCK_ID NUMBER,
    CONSTRAINT SECORDERS_STOCK_ID FOREIGN KEY(STOCK_ID) REFERENCES STOCK_INFO (STOCK_ID),
    ORDER_TYPE VARCHAR2(50),
    QUANTITY NUMBER,
    PRICE NUMBER,
    ORDER_STATUS VARCHAR2(50),
    ORDER_DATE DATE,
    ACCOUNT_ID NUMBER,
    CONSTRAINT SECORDERS_ACCOUNT_ID FOREIGN KEY(ACCOUNT_ID) REFERENCES ACCOUNT_INFO (ACCOUNT_ID),
    USER_ID NUMBER,
    CONSTRAINT SECORDERS_USER_ID FOREIGN KEY(USER_ID) REFERENCES CUSTOMER_INFO (USER_ID)
);

CREATE TABLE HOLDING (
    HOLDING_ID NUMBER PRIMARY KEY,
    ACCOUNT_ID NUMBER,
    CONSTRAINT HODLNGS_ACCOUNT_ID FOREIGN KEY(ACCOUNT_ID) REFERENCES ACCOUNT_INFO (ACCOUNT_ID),
    USER_ID NUMBER,
    CONSTRAINT HOLDINGS_USER_ID FOREIGN KEY(USER_ID) REFERENCES CUSTOMER_INFO (USER_ID),
    STOCK_ID NUMBER,
    CONSTRAINT HOLDINGS_STOCK_ID FOREIGN KEY(STOCK_ID) REFERENCES STOCK_INFO (STOCK_ID),
    QUANTITY NUMBER,
    AVERAGE_PRICE NUMBER,
    LAST_UPDATED DATE
);
```
