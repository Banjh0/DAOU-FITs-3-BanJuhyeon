package org.example.finalboardex.controller;

import org.example.finalboardex.service.BoardService;
import org.example.finalboardex.service.CommentService;
import org.example.finalboardex.service.LikeService;
import org.example.finalboardex.vo.BoardListVO;
import org.example.finalboardex.vo.BoardVO;

import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

@WebServlet(value = "/board")
public class BoardServlet extends HttpServlet {
    BoardService service = new BoardService();
    CommentService commentService = new CommentService();
    LikeService likeService = new LikeService();

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        String board_id = req.getParameter("board_id");
        String updateFlag = req.getParameter("updateFlag");
        if(board_id != null) {
            int boardId = Integer.parseInt(board_id);
            BoardVO board = service.selectBoardById(boardId);
            req.setAttribute("board", board);
            RequestDispatcher dispatcher = null;
            if("true".equals(updateFlag)) {
                dispatcher = req.getRequestDispatcher("/boardUpdate.jsp");
            } else {
                dispatcher = req.getRequestDispatcher("/boardDetail.jsp");
            }
            dispatcher.forward(req, resp);
        } else {
            List<BoardVO> boardList = null;
            boardList = service.selectAll();
            List<BoardListVO> boardListWithCnt = new ArrayList<>();

            for (BoardVO board : boardList) {
                int commentCount = commentService.getCommentCount(board.getBoard_id());
                int likeCount = likeService.getLikeCount(board.getBoard_id());
                boardListWithCnt.add(new BoardListVO(board, commentCount, likeCount));
            }
            req.setAttribute("boardList", boardListWithCnt);
            RequestDispatcher dispatcher = req.getRequestDispatcher("/boardList.jsp");
            dispatcher.forward(req, resp);
        }

    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");
        resp.setContentType("text/html;charset=UTF-8");
        String board_id = req.getParameter("boardId");
        String updateFlag = req.getParameter("updateFlag");
        if(board_id != null && !"true".equals(updateFlag)) {
            int boardId = Integer.parseInt(board_id);
            service.deleteBoard(boardId);
            resp.sendRedirect("/FinalBoardEx/board");
        } else {
            String title = req.getParameter("title");
            String content = req.getParameter("content");
            String userId = req.getParameter("user_id");
            BoardVO board = new BoardVO(userId, title, content);
            if("true".equals(updateFlag)) {
                int boardId = Integer.parseInt(board_id);
                service.updateBoard(boardId, board);
            } else {
                service.createBoard(board);
            }
            resp.sendRedirect("/FinalBoardEx/board");
        }
    }
}

package org.example.finalboardex.service;

import org.apache.ibatis.session.SqlSessionFactory;
import org.example.finalboardex.dao.LikeDAO;
import org.example.finalboardex.util.MyBatisSessionFactory;
import org.example.finalboardex.vo.LikeResponse;

import java.util.HashMap;
import java.util.Map;

public class LikeService {
    SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();

    LikeDAO dao = new LikeDAO(factory);

    public int getLikeCount(int boardId) {
        int likeCnt = 0;
        try {
            SqlSession session = factory.openSession();
            likeCnt = dao.getLikeCnt(session, boardId);
            session.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        return likeCnt;
    }

    public LikeResponse toggleLike(int boardId, String userId, boolean isLike) {
        SqlSession session = null;
        LikeResponse likeResponse = null;
        try {
            session = factory.openSession(false);
            if(!isLike) {
                likeResponse = dao.doLike(session, boardId, userId);
            } else {
                likeResponse = dao.unDoLike(session, boardId, userId);
            }
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }
        return likeResponse;
    }

    public boolean isLike(int boardId, String userId) {
        boolean isLike = false;
        int likeCnt = -1;
        try {
            SqlSession session = factory.openSession();
            likeCnt = dao.isLike(session, boardId, userId);
            session.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        if(likeCnt > 0) {
            isLike = true;
        }
        return isLike;
    }
}

            } else {

        }
    });

function addComment(boardId, userId) {
    let commentContent = $("#commentContent").val().trim();

    if(commentContent === "") {
        alert("댓글을 입력해주세요.");
        return;
    }

    $.ajax({
        url: "/FinalBoardEx/comment",
        type: "POST",
        data: {
            boardId: boardId,
            userId: userId,
            content: commentContent
        },
        success: function(data) {
            if(data.success) {
                let newComment = `<p id="comment_${data.commentId}"><strong>${userId}</strong> : ${commentContent} <button class="delete-comment" onclick="deleteComment(${data.commentId})" style="color:red; border:none; background:none; cursor:pointer;">X</button> </p>`;
                $("#commentList").append(newComment);
                $("#commentContent").val("");
            } else {
                alert("댓글 등록에 실패했습니다.");
            }
        },
        error: function() {
            alert("댓글 작성 요청에 실패했습니다.");
        }
    });
}

function loadComments(boardId, currentUserId) {
    $.ajax({
        async: false,
        url: "/FinalBoardEx/comment",
        type: "GET",
        data: {
            boardId: boardId
        },
        success: function (data) {
            $("#commentList").empty();
            if (data.length > 0) {
                data.forEach(comment => {
                    let deleteButton = "";
                    if (comment.userId === currentUserId) {
                        deleteButton = `<button class="delete-comment" onclick="deleteComment(${comment.commentId})" style="color:red; border:none; background:none; cursor:pointer;">X</button>`;
                    }
                    let commentHtml = `
                        <p id="comment_${comment.commentId}">
                            <strong>${comment.userId}</strong>: ${comment.content} 
<!--                             <span style="font-size: 12px; color: gray;">${comment.createdAt}</span>-->
                            ${deleteButton}
                        </p>`;
                    $("#commentList").append(commentHtml);
                });
            } else {
                $("#commentList").html("<p>댓글이 없습니다.</p>");
            }
        },
        error: function () {
            alert("댓글을 불러오는 중 오류가 발생했습니다.");
        }
    });
}

function deleteComment(commentId) {
    if (!confirm("댓글을 삭제하시겠습니까?")) return;

        url: "/FinalBoardEx/comment",

        success: function (data) {
            if (data.success) {
                alert("댓글이 삭제되었습니다.");
                $(`#comment_${commentId}`).remove();
            } else {
                alert("댓글 삭제에 실패했습니다.");
            }
        },
        error: function () {
