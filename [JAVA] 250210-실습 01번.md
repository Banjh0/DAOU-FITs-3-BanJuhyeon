// 책 제목에 대한 키워드를 입력받아서
// 책 제목에 해당 키워드를 포함하고 있는 책을 검색한 다음
// 해당 책에 대한 책 ISBN, 책 제목, 저자, 가격 정보를 출력하세요!

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.Scanner;

public class BookSearchCommand {
    public static void main(String[] args) {
        Connection con = null;
        try {
            // 1. JDBC Driver loading
            Class.forName("oracle.jdbc.driver.OracleDriver");
            
            // 2.데이터베이스 연결(Connection)
            String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
            String ID = "C##JAVA";
            String PW = "1234";
            con = DriverManager.getConnection(JDBC_URL, ID, PW);

            Scanner s = new Scanner(System.in);
            System.out.println("검색어를 입력하세요 => ");
            String keyword = s.nextLine();

            // 3. SQL구문 작성
/*
            String sql = "SELECT btitle, bauthor, bprice";
            sql += "FROM book WHERE btitle LIKE '%" + keyword + "%'";
*/
            // 위와 같이 작성해도 되긴 하지만, injection 문제와 보기에 좋지 않음.
            // -> IN Parameter를 이용해서 SQL query를 작성
            // ?는 나중에 채운다는 의미. 변할 수 있는 부분을 표현, 변수처리.
            String sql = "SELECT bisbn, btitle, bauthor, bprice";
            sql += " FROM book WHERE btitle LIKE ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            // 아직 SQL 문장이 완성 안되었기에 실행 안됨
            pstmt.setString(1, "%" + keyword + "%");
            // 여기서 1은 ? 중 1번째를 의미. 프로그래밍에성는 0부터지만, DB에서는 1부터.
            // SQL 쓴다고 안에 ' 넣은 것  아님 주의!

            // 4. SQL구문 실행
            ResultSet rs = pstmt.executeQuery();

            // 5. 결과처리
            while(rs.next()) {
                System.out.println(rs.getString("bisbn") + " " +
                        rs.getString("btitle") + " " +
                        rs.getString("bauthor") + " " +
                        rs.getString("bprice"));
            }

            // 6. 리소스 해제
            rs.close();
            pstmt.close();
            con.close();

        } catch(Exception e) {
            System.out.println("먼가 이상해요!!" + e.getMessage());
        }
    }
}
