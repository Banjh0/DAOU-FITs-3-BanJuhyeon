# TIL: Open API 이용하기

학습일자: 2025년 02월 20일

## 학습하는데 어떤 문제에 부딪혔는지

영화진흥위원회나 카카오의 Open API를 이용하게 되었다.

## 그 문제를 해결하기 위해 내가 시도 해본 것은

이전에 배운 내용을 다시 훑어보았다.

## 어떻게 그 문제를 해결 했는지

이전에 배운 내용과 같이 활용해보았다.

## 오늘 내가 새롭게 배운 것은 무엇인지

```jsx
function searchFunc() {
    let selectedDate = $("#selDate").val()
    let formattedDate = selectedDate.split("-").join("")
    $.ajax({
        async: true,
        url: "http://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json",
        type: "GET",
        data: {
            key: "",
            targetDt: formattedDate,
        },
        dataType: "json",

        success: function(result) {
            let dailyBoxOfficeList = result.boxOfficeResult.dailyBoxOfficeList;

            let imgUrl = "";

            $("table #result").empty()

            dailyBoxOfficeList.forEach(item => {
                $.ajax({
                    async: false, // true 하면 이미지 로드 전에 화면에 표시됨.
                    url: "https://dapi.kakao.com/v2/search/image",
                    type: "GET",
                    data: {
                        query: item.movieNm,
                    },
                    dataType: "json",
                    headers: {
                        Authorization: "KakaoAK ",
                    },

                    success: function(result) {
                        imgUrl = result.documents[0].image_url
                    },

                    error: function(result) {
                        console.log(result);
                        alert("이미지 불러오기에 실패했습니다")
                    }

                })

                let row = `
            <tr>
                <td>${item.rank}</td>
                <td><img src="${imgUrl}" width="150" height="200" /></td>
                <td>${item.movieNm}</td>
                <td>${item.audiAcc}</td>
                <td>${item.openDt}</td> 
                <td><input type="button" value="삭제" class="deleteBtn" onclick="deleteRow()"></td>
            </tr>
        `;
                $("table #result").append(row);
            });
        },

        error: function() {
            alert("영화 찾기에 실패하였습니다")
        }
    })
}
```

위 코드를 통해 화면상의 날짜를 정하면, 자동으로 함수가 실행되게 하였고, 영화의 정보들은 영화진흥위원회의 API를 이용해서 받아왔다. 다만, 영화 포스터는 제공되지 않아서 별도로 카카오의 Open API를 이용해 이미지 검색 기능을 이용했다. 이 때 이미지 URL이 호출되기 이전에 다른 내용들이 화면에 올라가는 경우가 있어 별도의 async 설정이 필요했다.
