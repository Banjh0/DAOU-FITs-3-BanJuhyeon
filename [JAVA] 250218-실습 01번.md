package org.example.boardex.controller;

import org.example.boardex.service.BoardService;
import org.example.boardex.vo.BoardVO;

import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

@WebServlet(value = "/board")
public class BoardServlet extends HttpServlet {
    BoardService service = new BoardService();

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        String board_id = req.getParameter("board_id");
        if(board_id != null) {
            int boardId = Integer.parseInt(board_id);
            BoardVO board = service.selectBoardById(boardId);
            req.setAttribute("board", board);
            RequestDispatcher dispatcher = req.getRequestDispatcher("/boardDetail.jsp");
            dispatcher.forward(req, resp);
        } else {
            List<BoardVO> boardList = null;
            boardList = service.selectAll();
            req.setAttribute("boardList", boardList);
            RequestDispatcher dispatcher = req.getRequestDispatcher("/boardList.jsp");
            dispatcher.forward(req, resp);
        }

    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");
        resp.setContentType("text/html;charset=UTF-8");
        String board_id = req.getParameter("boardId");
        if(board_id != null) {
            int boardId = Integer.parseInt(board_id);
            service.deleteBoard(boardId);
            resp.sendRedirect("/BoardEx/board");
        } else {
            String title = req.getParameter("title");
            String content = req.getParameter("content");
            String userId = req.getParameter("user_id");
            BoardVO board = new BoardVO(userId, title, content);
            service.createBoard(board);
            resp.sendRedirect("/BoardEx/board");
        }
    }
}

import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.io.PrintWriter;

@WebServlet(value="/login")
public class UserServlet extends HttpServlet {
    UserService service = new UserService();

    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");
        String id = req.getParameter("id");
        String pw = req.getParameter("pw");
        UserVO user = null;
        user = service.login(id, pw);
        resp.setContentType("text/html;charset=utf-8");
        if(user != null) { // 성공시
            HttpSession session = req.getSession();
            session.setAttribute("user", user);
            resp.sendRedirect("/BoardEx/board");
        } else { //실패시
            resp.sendRedirect("index.html");
        }
    }

    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        HttpSession session = req.getSession();
        UserVO user = (UserVO) session.getAttribute("user");

        if(user != null) {
            resp.sendRedirect("/BoardEx/board");
        } else {
            resp.sendRedirect("index.html");
        }
    }
}

package org.example.boardex.service;

import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.example.boardex.dao.BoardDAO;
import org.example.boardex.util.MyBatisSessionFactory;
import org.example.boardex.vo.BoardVO;

import java.util.List;

public class BoardService {
    SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();

    BoardDAO dao = new BoardDAO(factory);


    public List<BoardVO> selectAll() {
        List<BoardVO> boardList = null;
        try {
            SqlSession session = factory.openSession();
            boardList = dao.selectAll(session);
            session.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        return boardList;
    }

    public BoardVO selectBoardById(int boardId) {
        BoardVO board = null;
        try {
            SqlSession session = factory.openSession();
            board = dao.selectOne(session, boardId);
            session.close();
        } catch(Exception e) {
            e.printStackTrace();
        }
        return board;
    }

    public void createBoard(BoardVO board) {
        SqlSession session = null;
        try {
            session = factory.openSession(false);
            dao.createBoard(session, board);
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }
    }

    public void deleteBoard(int board_id) {
        SqlSession session = null;
        try {
            session = factory.openSession(false);
            dao.deleteBoard(session, board_id);
            session.commit();
        } catch(Exception e) {
            e.printStackTrace();
            session.rollback();
        } finally {
            session.close();
        }

    }
}

package org.example.boardex.util;

import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.Reader;

public class MyBatisSessionFactory {
    private static SqlSessionFactory sqlSessionFactory;

    static {
        try {
            String resource = "./SqlMapConfig.xml";
            Reader reader = Resources.getResourceAsReader(resource);

            if(sqlSessionFactory == null) {
                sqlSessionFactory =
                        new SqlSessionFactoryBuilder().build(reader);
            }

        } catch(Exception e) {
            e.printStackTrace();
        }
    }

    public static SqlSessionFactory getSqlSessionFactory() {
        return sqlSessionFactory;
    }
}
